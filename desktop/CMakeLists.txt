cmake_minimum_required(VERSION 3.20)
project(FlowGateDesktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(FLOWGATE_PRODUCT_NAME "FlowGate" CACHE STRING "Desktop display name")
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Test)
add_library(flowgate-ui window.cpp)
target_link_libraries(flowgate-ui PUBLIC Qt6::Widgets Qt6::Network)
target_compile_definitions(flowgate-ui PUBLIC FLOWGATE_PRODUCT_NAME="${FLOWGATE_PRODUCT_NAME}")
target_include_directories(flowgate-ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(flowgate-desktop main.cpp)
target_link_libraries(flowgate-desktop PRIVATE flowgate-ui)
add_executable(desktop-test test.cpp)
target_link_libraries(desktop-test PRIVATE flowgate-ui Qt6::Test)
